/* ======================= one-time setup ======================= */
function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.keys(TABS).forEach(name => {
    let sh = ss.getSheetByName(name);
    if (!sh) sh = ss.insertSheet(name);
    const head = TABS[name];
    sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground('#16324a').setFontColor('#ffffff');
    sh.setFrozenRows(1);
  });
  const set = ss.getSheetByName('Settings');
  const haveKeys = set.getLastRow() > 1 ? set.getRange(2, 1, set.getLastRow() - 1, 1).getValues().map(r => String(r[0]).trim()) : [];
  const missing = DEFAULT_SETTINGS.filter(r => haveKeys.indexOf(r[0]) < 0);   // safe to re-run: only adds settings that aren't there yet
  if (missing.length) set.getRange(set.getLastRow() + 1, 1, missing.length, 3).setNumberFormat('@').setValues(missing);
  const ns = ss.getSheetByName('No School Days');
  ns.getRange('A:B').setNumberFormat('@');
  if (ns.getLastRow() < 2) ns.getRange(2, 1, DEFAULT_NO_SCHOOL.length, 3).setValues(DEFAULT_NO_SCHOOL);
  const bells = ss.getSheetByName('Bells');
  if (bells.getLastRow() < 2) {
    bells.getRange(2, 1, 10, 4).setValues([
      ['08:30', '08:45', 'Yard supervision', ''],
      ['08:45', '08:50', 'Entry bell', 'Safe Arrival (AM)'],
      ['08:50', '10:30', 'Learning Block 1', ''],
      ['10:30', '10:50', 'Nutrition Break 1', ''],
      ['10:50', '11:10', 'Recess 1', ''],
      ['11:10', '12:50', 'Learning Block 2', 'After recess / break'],
      ['12:50', '13:10', 'Nutrition Break 2', ''],
      ['13:10', '13:30', 'Recess 2', ''],
      ['13:30', '15:10', 'Learning Block 3', 'Safe Arrival (PM)'],
      ['15:10', '15:25', 'Dismissal', '']
    ]);
  }
  ['Bells', 'Duty', 'Schedules'].forEach(n => ss.getSheetByName(n).getRange('A:D').setNumberFormat('@'));
  ss.getSheetByName('Duty Changes').getRange('A:A').setNumberFormat('yyyy-mm-dd');
  ss.getSheetByName('Bells').getRange('D2:D200').setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(CHECKPOINTS, true).setAllowInvalid(false).build());
  ss.getSheetByName('Staff').getRange('C2:C500').setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(ROLES, true).setAllowInvalid(true).build());
  ss.getSheetByName('Duty').getRange('A2:A500').setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(['Day 1', 'Day 2', 'Day 3', 'Day 4', 'Day 5', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Every day'], true).setAllowInvalid(true).build());
  const props = PropertiesService.getScriptProperties();
  if (!props.getProperty('SESSION_SECRET')) props.setProperty('SESSION_SECRET', Utilities.getUuid() + Utilities.getUuid());
  const s1 = ss.getSheetByName('Sheet1'); if (s1 && s1.getLastRow() === 0 && ss.getSheets().length > 1) ss.deleteSheet(s1);
  try { SpreadsheetApp.getUi().alert('WPS Staff Hub tabs are ready. Add staff, students and schedules, then Deploy (or redeploy a New version) as a web app.'); } catch (e) {}
}

/* ======================= web endpoints ======================= */
function doGet(e) {
  const p = e.parameter || {};
  try {
    if (p.action === 'login') return json_(login_(p.token));
    const me = auth_(p.session);
    switch (p.action) {
      case 'all':  return json_(all_(me));
      case 'live': return json_(live_(me));
      case 'week': return json_(week_(me, p.homeroom, p.monday));
      case 'photos': return json_(photos_(me, String(p.keys || '').split('|').filter(String).slice(0, 60)));
      default:     return json_({ error: 'unknown_action' });
    }
  } catch (err) { return json_({ error: err.message || String(err) }); }
}
function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const me = auth_(body.session);
    if (body.action === 'attendance') return json_(saveAttendance_(me, body.entries || []));
    if (body.action === 'admin') return json_(admin_(me, body.ops || []));
    return json_({ error: 'unknown_action' });
  } catch (err) { return json_({ error: err.message || String(err) }); }
}
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

/* ======================= auth ======================= */
function login_(token) {
  if (!token) throw new Error('no_token');
  const res = UrlFetchApp.fetch('https://oauth2.googleapis.com/tokeninfo?access_token=' + encodeURIComponent(token), { muteHttpExceptions: true });
  if (res.getResponseCode() !== 200) throw new Error('bad_token');
  const info = JSON.parse(res.getContentText());
  const settings = settings_();
  const clientId = String(settings['OAuth Client ID'] || '').trim();
  if (clientId && info.aud !== clientId && info.azp !== clientId) throw new Error('bad_token');
  if (String(info.email_verified) !== 'true') throw new Error('bad_token');
  const email = String(info.email || '').toLowerCase();
  const me = whoIs_(email, settings);
  if (!me) throw new Error('not_staff');
  const days = Number(settings['Session Days']) || 30;
  return { session: sign_(email, Date.now() + days * 86400000), me: me };
}
function whoIs_(email, settings) {
  const staff = rows_('Staff');
  const mine = staff.filter(r => String(r['Email']).toLowerCase().trim() === email);
  const appAdmin = appAdmins_(settings).indexOf(email) > -1;
  if (mine.length) {
    return { email: email, name: mine[0]['Name'] || email, role: mine[0]['Role'] || 'Teacher', homerooms: mine.map(r => r['Homeroom']).filter(String), appAdmin: appAdmin };
  }
  const domain = String(settings['Allowed Domain'] || '').toLowerCase().replace(/^@/, '');
  const guests = String(settings['Allow Board Guests']).toUpperCase() === 'TRUE';
  if (guests && domain && email.endsWith('@' + domain)) return { email: email, name: email.split('@')[0], role: 'Guest', homerooms: [] };
  return null;
}
function sign_(email, exp) {
  const payload = Utilities.base64EncodeWebSafe(email + '|' + exp);
  return payload + '.' + hmac_(payload);
}
function hmac_(s) {
  const secret = PropertiesService.getScriptProperties().getProperty('SESSION_SECRET');
  if (!secret) throw new Error('Run setupSheet() first');
  return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(s, secret));
}
function auth_(session) {
  if (!session || session.indexOf('.') < 0) throw new Error('session');
  const parts = session.split('.');
  if (hmac_(parts[0]) !== parts[1]) throw new Error('session');
  const raw = Utilities.newBlob(Utilities.base64DecodeWebSafe(parts[0])).getDataAsString().split('|');
  if (Number(raw[1]) < Date.now()) throw new Error('session');
  const me = whoIs_(raw[0], settings_());   // re-checked every request, so removing someone from Staff locks them out
  if (!me) throw new Error('session');
  return me;
}
const isOffice_ = me => /admin|office|principal/i.test(me.role || '');
/* Admin access comes only from the "App Admins" setting, not from the Staff tab's Role column. */
const isAdmin_ = me => !!me.appAdmin;
function appAdmins_(settings) { return String(settings['App Admins'] || '').toLowerCase().split(/[,;\s]+/).filter(String); }

/* ======================= data ======================= */
function all_(me) {
  const s = settings_();
  return {
    me: me,
    settings: { 'School Name': s['School Name'], 'Short Name': s['Short Name'], 'Office Extension': s['Office Extension'],
                'Cycle Days': s['Cycle Days'], 'Cycle Anchor Date': s['Cycle Anchor Date'], 'Cycle Anchor Day': s['Cycle Anchor Day'],
                'Show Attendance': s['Show Attendance'], 'Allow Board Guests': s['Allow Board Guests'], 'Guest Photos': s['Guest Photos'] },
    sheetUrl: isAdmin_(me) ? SpreadsheetApp.getActiveSpreadsheet().getUrl() : '',
    noSchool: rows_('No School Days'),
    staff: rows_('Staff'), students: rows_('Students'), bells: rows_('Bells'),
    duty: rows_('Duty'), dutyChanges: rows_('Duty Changes').filter(r => String(r['Date']) >= today_()),
    schedules: rows_('Schedules')
  };
}
function live_(me) {
  if (!isOffice_(me)) throw new Error('office_only');
  const t = today_();
  return { rows: rows_('Attendance').filter(r => r['Date'] === t && r['Status'] !== 'Deleted') };
}
function week_(me, homeroom, monday) {
  const start = monday, endD = new Date(monday + 'T12:00:00'); endD.setDate(endD.getDate() + 4);
  const end = Utilities.formatDate(endD, tz_(), 'yyyy-MM-dd');
  const rows = rows_('Attendance').filter(r => r['Homeroom'] === homeroom && r['Date'] >= start && r['Date'] <= end && r['Status'] !== 'Deleted');
  return { rows: rows.map(r => ({ id: r['Entry ID'], date: r['Date'], time: r['Time'], count: Number(r['Count']), checkpoint: r['Checkpoint'], status: String(r['Status']).toLowerCase(), note: r['Note'] })) };
}
/** Student photos live only in the Photos tab of this (private) Sheet and are sent only to signed-in staff. */
function photos_(me, keys) {
  if (!keys.length) return { photos: {} };
  if (me.role === 'Guest' && String(settings_()['Guest Photos']).toUpperCase() === 'FALSE') return { photos: {}, hidden: true };
  const cache = CacheService.getScriptCache(), hit = cache.getAll(keys.map(k => 'ph:' + k)), out = {};
  let missing = keys.filter(k => !(('ph:' + k) in hit));
  keys.forEach(k => { if (hit['ph:' + k] != null) out[k] = hit['ph:' + k] || null; });
  if (missing.length) {
    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Photos');
    const vals = sh && sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues() : [];
    const all = {}; vals.forEach(r => { if (r[0]) all[String(r[0]).trim()] = String(r[1] || ''); });
    const put = {};   // cache every photo for 6 hours so the next class loads instantly
    Object.keys(all).forEach(k => { if (all[k].length < 95000) put['ph:' + k] = all[k]; });
    try { const ks = Object.keys(put); for (let i = 0; i < ks.length; i += 100) { const chunk = {}; ks.slice(i, i + 100).forEach(k => chunk[k] = put[k]); cache.putAll(chunk, 21600); } } catch (e) {}
    missing.forEach(k => { out[k] = all[k] || null; });
  }
  return { photos: out };
}

function saveAttendance_(me, entries) {
  const lock = LockService.getScriptLock(); lock.waitLock(15000);
  try {
    const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Attendance');
    const last = sh.getLastRow();
    const ids = last > 1 ? sh.getRange(2, 12, last - 1, 1).getValues().map(r => String(r[0])) : [];
    const add = [];
    entries.forEach(e => {
      if (e.op === 'add' && ids.indexOf(String(e.id)) < 0) {
        const status = String(e.status || 'ok');
        add.push([new Date(), e.date, e.time, e.homeroom, me.email, me.name, e.checkpoint, Number(e.count), Number(e.expected),
          status.charAt(0).toUpperCase() + status.slice(1), e.note || '', String(e.id)]);
        ids.push(String(e.id));
      }
      if (e.op === 'delete') {
        const i = ids.indexOf(String(e.id));
        if (i > -1 && i < last - 1) sh.getRange(i + 2, 10).setValue('Deleted');
        else if (i > -1) add.forEach(r => { if (r[11] === String(e.id)) r[9] = 'Deleted'; });
      }
    });
    if (add.length) sh.getRange(sh.getLastRow() + 1, 1, add.length, add[0].length).setValues(add);
    return { ok: true, saved: add.length };
  } finally { lock.releaseLock(); }
}

/* ======================= helpers ======================= */
function tz_() { return SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone() || 'America/Toronto'; }
function today_() { return Utilities.formatDate(new Date(), tz_(), 'yyyy-MM-dd'); }
function settings_() {
  const out = {};
  rows_('Settings').forEach(r => { if (r['Setting']) out[String(r['Setting']).trim()] = r['Value']; });
  return out;
}
/** Reads a tab into objects keyed by header. Times become "HH:mm", dates "yyyy-MM-dd". */
/* ======================= admin (Admin role only) ======================= */
const ADMIN_SETTINGS = ['Show Attendance', 'Allow Board Guests', 'Guest Photos', 'Office Extension', 'Cycle Anchor Date', 'Cycle Anchor Day'];
function admin_(me, ops) {
  if (!isAdmin_(me)) throw new Error('admin_only');
  const ss = SpreadsheetApp.getActiveSpreadsheet(), lock = LockService.getScriptLock(); lock.waitLock(15000);
  try {
    ops.forEach(op => {
      if (op.type === 'setting') {
        if (ADMIN_SETTINGS.indexOf(op.key) < 0) throw new Error('bad_setting');
        setSetting_(ss, op.key, String(op.value));
      } else if (op.type === 'addNoSchool') {
        appendText_(ss, 'No School Days', [op.start, op.end || '', op.reason || 'No school']);
      } else if (op.type === 'removeNoSchool') {
        deleteWhere_(ss, 'No School Days', r => r['Start Date'] === op.start && String(r['Reason']) === String(op.reason || ''));
      } else if (op.type === 'addDutyChange') {
        appendText_(ss, 'Duty Changes', [op.date, op.block || '', op.location, op.coveredBy, op.replacing || '', op.note || '']);
      } else if (op.type === 'removeDutyChange') {
        deleteWhere_(ss, 'Duty Changes', r => r['Date'] === op.date && String(r['Location']) === String(op.location) && String(r['Time Block']) === String(op.block || ''));
      } else throw new Error('bad_op');
    });
    SpreadsheetApp.flush();
  } finally { lock.releaseLock(); }
  return { ok: true, data: all_(me) };
}
function setSetting_(ss, key, value) {
  const sh = ss.getSheetByName('Settings'), last = sh.getLastRow();
  const keys = last > 1 ? sh.getRange(2, 1, last - 1, 1).getValues().map(r => String(r[0]).trim()) : [];
  const i = keys.indexOf(key);
  if (i > -1) sh.getRange(i + 2, 2).setNumberFormat('@').setValue(value);
  else sh.getRange(last + 1, 1, 1, 3).setNumberFormat('@').setValues([[key, value, '']]);
}
function appendText_(ss, name, row) {
  const sh = ss.getSheetByName(name), r = sh.getLastRow() + 1;
  sh.getRange(r, 1, 1, row.length).setNumberFormat('@').setValues([row.map(String)]);
}
function deleteWhere_(ss, name, match) {
  const sh = ss.getSheetByName(name); if (!sh || sh.getLastRow() < 2) return;
  const values = sh.getDataRange().getValues(), head = values.shift().map(h => String(h).trim()), tz = tz_();
  for (let i = values.length - 1; i >= 0; i--) if (match(rowObj_(head, values[i], tz))) sh.deleteRow(i + 2);
}

function rowObj_(head, r, tz) {
  const o = {};
  head.forEach((h, i) => {
    let v = r[i];
    if (v instanceof Date) v = v.getFullYear() < 1901 ? Utilities.formatDate(v, tz, 'HH:mm') : (h === 'Logged At' ? Utilities.formatDate(v, tz, 'yyyy-MM-dd HH:mm') : Utilities.formatDate(v, tz, 'yyyy-MM-dd'));
    o[h] = typeof v === 'string' ? v.trim() : v;
  });
  return o;
}
function rows_(name) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
  if (!sh || sh.getLastRow() < 2) return [];
  const values = sh.getDataRange().getValues();
  const head = values.shift().map(h => String(h).trim());
  const tz = tz_();
  return values.filter(r => r.some(v => v !== '' && v !== null)).map(r => rowObj_(head, r, tz));
}
